-- PARTE 2 ---


1. CLIENTES
Entidade: Cadastro dos clientes da oficina, armazenando informações pessoais e de contato.

id_cliente: Identificador único do cliente.
nome: Nome completo do cliente.
cpf: Número do CPF do cliente.
telefone: Número de telefone do cliente.
email: Endereço de e-mail do cliente.
endereco: Endereço residencial do cliente.
2. VEICULOS
Entidade: Cadastro dos veículos dos clientes atendidos pela oficina.

id_veiculo: Identificador único do veículo.
placa: Placa de identificação do veículo.
marca: Marca do veículo.
modelo: Modelo do veículo.
ano: Ano de fabricação do veículo.
cor: Cor predominante do veículo.
3. FUNCIONARIOS
Entidade: Cadastro dos funcionários que trabalham na oficina e realizam os serviços.

id_funcionario: Identificador único do funcionário.
nome: Nome completo do funcionário.
cpf: Número do CPF do funcionário.
cargo: Função exercida pelo funcionário na oficina.
telefone: Número de telefone do funcionário.
salario: Valor do salário recebido pelo funcionário.
4. SERVICOS
Entidade: Cadastro dos serviços oferecidos pela oficina para manutenção e reparo dos veículos.

id_servico: Identificador único do serviço.
nome: Nome do serviço oferecido pela oficina.
descricao: Descrição das atividades realizadas no serviço.
preco: Valor cobrado pelo serviço.
tempo_estimado: Tempo estimado para realização do serviço em minutos.
status: Situação atual do serviço no cadastro.
5. ORDENS_SERVICO
Entidade: Registro das ordens de serviço abertas para manutenção e reparo dos veículos.

id_ordem_servico: Identificador único da ordem de serviço.
data_abertura: Data em que a ordem de serviço foi aberta.
data_fechamento: Data em que a ordem de serviço foi finalizada.
valor_total: Valor total dos serviços e peças utilizados na ordem.
status: Situação atual da ordem de serviço.
observacao: Informações adicionais sobre a ordem de serviço.
6. PECAS
Entidade: Cadastro das peças utilizadas nos serviços de manutenção e reparo dos veículos.

id_peca: Identificador único da peça.
nome: Nome da peça utilizada pela oficina.
descricao: Descrição das características da peça.
preco: Valor de venda da peça.
quantidade: Quantidade disponível da peça.
fabricante: Nome do fabricante da peça.
7. FORNECEDORES
Entidade: Cadastro dos fornecedores responsáveis pelo fornecimento de peças e materiais para a oficina.

id_fornecedor: Identificador único do fornecedor.
nome: Nome ou razão social do fornecedor.
cnpj: Número do CNPJ do fornecedor.
telefone: Número de telefone do fornecedor.
email: Endereço de e-mail do fornecedor.
endereco: Endereço comercial do fornecedor.
8. ESTOQUE
Entidade: Controle das peças e materiais disponíveis no estoque da oficina.

id_estoque: Identificador único do registro de estoque.
quantidade: Quantidade atual de itens disponíveis no estoque.
estoque_minimo: Quantidade mínima que deve ser mantida em estoque.
data_entrada: Data de entrada do item no estoque.
data_atualizacao: Data da última atualização do estoque.
localizacao: Local onde o item está armazenado na oficina.
9. PAGAMENTOS
Entidade: Registro dos pagamentos realizados pelos clientes pelos serviços e peças utilizados.

id_pagamento: Identificador único do pagamento.
valor: Valor pago pelo cliente.
data_pagamento: Data em que o pagamento foi realizado.
forma_pagamento: Forma utilizada para realizar o pagamento.
status: Situação atual do pagamento.
observacao: Informações adicionais relacionadas ao pagamento.
10. AGENDAMENTOS
Entidade: Registro dos agendamentos realizados pelos clientes para atendimento na oficina.

id_agendamento: Identificador único do agendamento.
data_agendamento: Data marcada para o atendimento.
hora_agendamento: Horário marcado para o atendimento.
motivo: Motivo informado pelo cliente para o atendimento.
status: Situação atual do agendamento.
observacao: Informações adicionais sobre o agendamento.

-- ============================================
-- BANCO DE DADOS - OFICINA
-- ============================================

CREATE DATABASE oficina;

USE oficina;


-- ============================================
-- 1. TABELA CLIENTES
-- ============================================

CREATE TABLE clientes (
    id_cliente INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(60) NOT NULL,
    cpf VARCHAR(14) NOT NULL UNIQUE,
    telefone VARCHAR(20),
    email VARCHAR(60),
    endereco VARCHAR(100)
);


-- ============================================
-- 2. TABELA MARCAS
-- ============================================

CREATE TABLE marcas (
    id_marca INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(60) NOT NULL
);


-- ============================================
-- 3. TABELA MODELOS
-- ============================================

CREATE TABLE modelos (
    id_modelo INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(60) NOT NULL,
    id_marca INT NOT NULL,

    CONSTRAINT fk_modelos_marcas
        FOREIGN KEY (id_marca)
        REFERENCES marcas(id_marca)
);


-- ============================================
-- 4. TABELA VEICULOS
-- ============================================

CREATE TABLE veiculos (
    id_veiculo INT AUTO_INCREMENT PRIMARY KEY,
    placa VARCHAR(10) NOT NULL UNIQUE,
    ano INT,
    cor VARCHAR(30),
    id_cliente INT NOT NULL,
    id_modelo INT NOT NULL,

    CONSTRAINT fk_veiculos_clientes
        FOREIGN KEY (id_cliente)
        REFERENCES clientes(id_cliente),

    CONSTRAINT fk_veiculos_modelos
        FOREIGN KEY (id_modelo)
        REFERENCES modelos(id_modelo)
);


-- ============================================
-- 5. TABELA FUNCIONARIOS
-- ============================================

CREATE TABLE funcionarios (
    id_funcionario INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(60) NOT NULL,
    cpf VARCHAR(14) NOT NULL UNIQUE,
    telefone VARCHAR(20),
    cargo VARCHAR(50),
    salario DECIMAL(10,2)
);


-- ============================================
-- 6. TABELA SERVICOS
-- ============================================

CREATE TABLE servicos (
    id_servico INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(60) NOT NULL,
    descricao VARCHAR(150),
    valor DECIMAL(10,2) NOT NULL
);


-- ============================================
-- 7. TABELA FORNECEDORES
-- ============================================

CREATE TABLE fornecedores (
    id_fornecedor INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(60) NOT NULL,
    cnpj VARCHAR(18) UNIQUE,
    telefone VARCHAR(20),
    email VARCHAR(60),
    endereco VARCHAR(100)
);


-- ============================================
-- 8. TABELA PECAS
-- ============================================

CREATE TABLE pecas (
    id_peca INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(60) NOT NULL,
    descricao VARCHAR(150),
    valor DECIMAL(10,2) NOT NULL,
    estoque INT DEFAULT 0,
    id_fornecedor INT NOT NULL,

    CONSTRAINT fk_pecas_fornecedores
        FOREIGN KEY (id_fornecedor)
        REFERENCES fornecedores(id_fornecedor)
);


-- ============================================
-- 9. TABELA ORDENS DE SERVICO
-- ============================================

CREATE TABLE ordens_servico (
    id_ordem INT AUTO_INCREMENT PRIMARY KEY,
    data_abertura DATE NOT NULL,
    data_fechamento DATE,
    descricao VARCHAR(200),
    status VARCHAR(30) NOT NULL,
    valor_total DECIMAL(10,2) DEFAULT 0,
    id_cliente INT NOT NULL,
    id_veiculo INT NOT NULL,
    id_funcionario INT NOT NULL,

    CONSTRAINT fk_ordens_clientes
        FOREIGN KEY (id_cliente)
        REFERENCES clientes(id_cliente),

    CONSTRAINT fk_ordens_veiculos
        FOREIGN KEY (id_veiculo)
        REFERENCES veiculos(id_veiculo),

    CONSTRAINT fk_ordens_funcionarios
        FOREIGN KEY (id_funcionario)
        REFERENCES funcionarios(id_funcionario)
);


-- ============================================
-- 10. TABELA PAGAMENTOS
-- ============================================

CREATE TABLE pagamentos (
    id_pagamento INT AUTO_INCREMENT PRIMARY KEY,
    data_pagamento DATE NOT NULL,
    valor DECIMAL(10,2) NOT NULL,
    forma_pagamento VARCHAR(30) NOT NULL,
    status VARCHAR(30) NOT NULL,
    id_ordem INT NOT NULL,

    CONSTRAINT fk_pagamentos_ordens
        FOREIGN KEY (id_ordem)
        REFERENCES ordens_servico(id_ordem)
);


-- ============================================
-- 11. RELACIONAMENTO ORDEM x SERVIÇOS
-- ============================================

CREATE TABLE ordem_servico_servicos (
    id_ordem INT NOT NULL,
    id_servico INT NOT NULL,
    quantidade INT DEFAULT 1,
    valor DECIMAL(10,2) NOT NULL,

    PRIMARY KEY (id_ordem, id_servico),

    CONSTRAINT fk_oss_ordem
        FOREIGN KEY (id_ordem)
        REFERENCES ordens_servico(id_ordem),

    CONSTRAINT fk_oss_servico
        FOREIGN KEY (id_servico)
        REFERENCES servicos(id_servico)
);


-- ============================================
-- 12. RELACIONAMENTO ORDEM x PEÇAS
-- ============================================

CREATE TABLE ordem_servico_pecas (
    id_ordem INT NOT NULL,
    id_peca INT NOT NULL,
    quantidade INT DEFAULT 1,
    valor DECIMAL(10,2) NOT NULL,

    PRIMARY KEY (id_ordem, id_peca),

    CONSTRAINT fk_osp_ordem
        FOREIGN KEY (id_ordem)
        REFERENCES ordens_servico(id_ordem),

    CONSTRAINT fk_osp_peca
        FOREIGN KEY (id_peca)
        REFERENCES pecas(id_peca)

Relacionamento	Descrição
Clientes → Veículos	Um cliente pode possuir vários veículos, mas cada veículo pertence a um único cliente.
Marcas → Modelos	Uma marca pode possuir vários modelos, mas cada modelo pertence a uma única marca.
Modelos → Veículos	Um modelo pode estar associado a vários veículos, mas cada veículo possui um único modelo.
Clientes → Ordens de Serviço	Um cliente pode possuir várias ordens de serviço, mas cada ordem pertence a um único cliente.
Veículos → Ordens de Serviço	Um veículo pode possuir várias ordens de serviço, mas cada ordem está relacionada a um único veículo.
Funcionários → Ordens de Serviço	Um funcionário pode ser responsável por várias ordens de serviço, mas cada ordem possui um funcionário responsável.
Fornecedores → Peças	Um fornecedor pode fornecer várias peças, mas cada peça está associada a um fornecedor.
Ordens de Serviço → Pagamentos	Uma ordem de serviço pode possuir um ou vários pagamentos, e cada pagamento pertence a uma ordem de serviço.
Ordens de Serviço → Serviços	Uma ordem de serviço pode possuir vários serviços e um serviço pode estar em várias ordens de serviço.
Ordens de Serviço → Peças	Uma ordem de serviço pode utilizar várias peças e uma peça pode ser utilizada em várias ordens de serviço.
