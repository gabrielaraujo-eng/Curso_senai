CLIENTES
id_cliente | INT(11) | PRIMARY KEY | NOT NULL | AUTO_INCREMENT
nome | VARCHAR(100) | NOT NULL
cpf | VARCHAR(14) | NOT NULL
telefone | VARCHAR(15) | NOT NULL
email | VARCHAR(150) | NOT NULL
endereco | VARCHAR(200) | NOT NULL

FUNCIONARIOS
id_funcionario | INT(11) | PRIMARY KEY | NOT NULL | AUTO_INCREMENT
nome | VARCHAR(100) | NOT NULL
cpf | VARCHAR(14) | NOT NULL
cargo | VARCHAR(50) | NOT NULL
salario | DECIMAL(10,2) | NOT NULL | DEFAULT 0.00
telefone | VARCHAR(15) | NOT NULL

PRODUTOS
id_produto | INT(11) | PRIMARY KEY | NOT NULL | AUTO_INCREMENT
nome | VARCHAR(100) | NOT NULL
descricao | VARCHAR(255)
preco | DECIMAL(10,2) | NOT NULL | DEFAULT 0.00
categoria | VARCHAR(50) | NOT NULL
validade | DATE

PEDIDOS
id_pedido | INT(11) | PRIMARY KEY | NOT NULL | AUTO_INCREMENT
data_pedido | DATE | NOT NULL | DEFAULT CURRENT_DATE
hora_pedido | TIME | NOT NULL | DEFAULT CURRENT_TIME
valor_total | DECIMAL(10,2) | NOT NULL | DEFAULT 0.00
status | VARCHAR(30) | NOT NULL | DEFAULT 'Pendente'
tipo_pedido | VARCHAR(30) | NOT NULL | DEFAULT 'Local'

PAGAMENTOS
id_pagamento | INT(11) | PRIMARY KEY | NOT NULL | AUTO_INCREMENT
valor | DECIMAL(10,2) | NOT NULL | DEFAULT 0.00
data_pagamento | DATE | NOT NULL | DEFAULT CURRENT_DATE
forma_pagamento | VARCHAR(30) | NOT NULL | DEFAULT 'Dinheiro'
status | VARCHAR(30) | NOT NULL | DEFAULT 'Pendente'
troco | DECIMAL(10,2) | NOT NULL | DEFAULT 0.00

FORNECEDORES
id_fornecedor | INT(11) | PRIMARY KEY | NOT NULL | AUTO_INCREMENT
nome | VARCHAR(100) | NOT NULL
cnpj | VARCHAR(18) | NOT NULL
telefone | VARCHAR(15) | NOT NULL
email | VARCHAR(150) | NOT NULL
endereco | VARCHAR(200) | NOT NULL

ESTOQUE
id_estoque | INT(11) | PRIMARY KEY | NOT NULL | AUTO_INCREMENT
quantidade | INT(11) | NOT NULL | DEFAULT 0
estoque_minimo | INT(11) | NOT NULL | DEFAULT 0
data_entrada | DATE | NOT NULL | DEFAULT CURRENT_DATE
data_atualizacao | DATE | NOT NULL | DEFAULT CURRENT_DATE
localizacao | VARCHAR(100)

CATEGORIAS
id_categoria | INT(11) | PRIMARY KEY | NOT NULL | AUTO_INCREMENT
nome | VARCHAR(50) | NOT NULL
descricao | VARCHAR(255)
tipo | VARCHAR(50) | NOT NULL
status | VARCHAR(20) | NOT NULL | DEFAULT 'Ativo'
observacao | VARCHAR(255)

ENTREGAS
id_entrega | INT(11) | PRIMARY KEY | NOT NULL | AUTO_INCREMENT
endereco | VARCHAR(200) | NOT NULL
data_entrega | DATE | NOT NULL | DEFAULT CURRENT_DATE
hora_entrega | TIME
status | VARCHAR(30) | NOT NULL | DEFAULT 'Pendente'
taxa_entrega | DECIMAL(10,2) | NOT NULL | DEFAULT 0.00

AVALIACOES
id_avaliacao | INT(11) | PRIMARY KEY | NOT NULL | AUTO_INCREMENT
nota | INT(1) | NOT NULL | DEFAULT 5
comentario | VARCHAR(500)
data_avaliacao | DATE | NOT NULL | DEFAULT CURRENT_DATE
status | VARCHAR(20) | NOT NULL | DEFAULT 'Ativo'
resposta | VARCHAR(500)
